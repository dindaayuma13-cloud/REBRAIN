<script>
    function appLogin() {
        const nip = document.getElementById('nip-input').value;
        if(!nip) {
            alert('Mohon masukkan NIP pegawai terlebih dahulu!');
            return;
        }
        
        // Sembunyikan login, tampilkan header, home screen, dan bottom nav
        document.getElementById('login-screen').classList.remove('active');
        document.getElementById('main-header').style.display = 'flex';
        document.getElementById('screen-home').classList.add('active');
        document.getElementById('main-nav').style.display = 'flex';
        
        document.getElementById('user-nip-badge').innerText = 'NIP: ' + nip;
    }

    function switchTab(screenId, element) {
        // Hilangkan active dari semua screen
        document.querySelectorAll('.screen').forEach(screen => {
            if(screen.id !== 'login-screen') screen.classList.remove('active');
        });

        // Tampilkan screen yang dipilih
        document.getElementById('screen-' + screenId).classList.add('active');

        // Atur navigasi aktif
        document.querySelectorAll('.nav-item').forEach(item => {
            item.classList.remove('active');
        });
        element.classList.add('active');
    }

    function alertMemutarVideo() {
        alert('Memuat pemutar video studi kasus lapangan dan panduan praktis dari senior...');
    }
</script>
